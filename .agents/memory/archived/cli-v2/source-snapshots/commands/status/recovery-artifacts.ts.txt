export type RecoveryArtifactKind = (typeof RecoveryArtifactKind)[keyof typeof RecoveryArtifactKind];

export type KnownRecoveryArtifact =

      readonly suffix: typeof GitlessBackupSuffix;
    }
  | {

      readonly suffix: typeof OpenForgeResidualSuffix.preparedReplacement;
    }

      readonly suffix: typeof OpenForgeResidualSuffix.preservedPrevious;
    };
